    <script>
        const canvas = document.getElementById('canvas');
        const ctx = canvas.getContext('2d');
        const workspace = document.getElementById('workspace');
        const emptyState = document.getElementById('empty-state');

        let currentImage = null; 
        let placedElements = []; 
        let activeElementSource = null; 
        let activeElementType = null;

        let isMeasuring = false;
        let measureStart = null;
        let measureEnd = null;

        document.getElementById('btn-new-project').addEventListener('click', () => {
            document.getElementById('upload-schema').click();
        });

        document.getElementById('upload-schema').addEventListener('change', function(e) {
            const file = e.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(event) {
                const img = new Image();
                img.onload = function() {
                    canvas.width = img.width;
                    canvas.height = img.height;
                    currentImage = img;
                    placedElements = [];
                    measureStart = null;
                    measureEnd = null;
                    
                    canvas.style.display = 'block';
                    emptyState.style.display = 'none';
                    drawCanvas();
                };
                img.src = event.target.result;
            };
            reader.readAsDataURL(file);
        });

        document.getElementById('btn-add-custom').addEventListener('click', () => {
            document.getElementById('upload-custom-element').click();
        });

        document.getElementById('upload-custom-element').addEventListener('change', function(e) {
            const file = e.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(event) {
                const grid = document.getElementById('custom-elements-grid');
                const newBtn = document.createElement('div');
                newBtn.className = 'sign-btn';
                
                const imgSrc = event.target.result;
                newBtn.onclick = function() { selectCustomSign(newBtn, imgSrc); };
                
                newBtn.innerHTML = `
                    <img src="${imgSrc}" alt="Elementas">
                    <span class="sign-label">${file.name.substring(0, 10)}...</span>
                `;
                grid.appendChild(newBtn);
            };
            reader.readAsDataURL(file);
        });

        function selectSign(buttonElement, type) {
            cancelActiveTools();
            document.querySelectorAll('.sign-btn').forEach(btn => btn.classList.remove('active'));
            buttonElement.classList.add('active');
            
            const imgEl = buttonElement.querySelector('img');
            activeElementSource = imgEl;
            activeElementType = type;
        }

        function selectCustomSign(buttonElement, imgSrc) {
            cancelActiveTools();
            document.querySelectorAll('.sign-btn').forEach(btn => btn.classList.remove('active'));
            buttonElement.classList.add('active');
            
            const img = new Image();
            img.src = imgSrc;
            activeElementSource = img;
            activeElementType = 'custom';
        }

        function cancelActiveTools() {
            isMeasuring = false;
            document.getElementById('btn-measure').classList.remove('active-tool');
            activeElementSource = null;
            activeElementType = null;
            document.querySelectorAll('.sign-btn').forEach(btn => btn.classList.remove('active'));
        }

        document.getElementById('btn-measure').addEventListener('click', () => {
            if (!currentImage) { alert("Pirmiausia įkelkite schemą."); return; }
            const btn = document.getElementById('btn-measure');
            isMeasuring = !isMeasuring;
            
            if (isMeasuring) {
                cancelActiveTools();
                isMeasuring = true;
                btn.classList.add('active-tool');
                measureStart = null;
                measureEnd = null;
            } else {
                btn.classList.remove('active-tool');
            }
            drawCanvas();
        });

        canvas.addEventListener('click', function(e) {
            if (!currentImage) return;

            const rect = canvas.getBoundingClientRect();
            const scaleX = canvas.width / rect.width;
            const scaleY = canvas.height / rect.height;
            const x = (e.clientX - rect.left) * scaleX
